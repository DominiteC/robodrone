十、滤波器实现代码示例 (C语言)
------------------------------

// 二阶IIR陷波滤波器实现
typedef struct {
    float b0, b1, b2;  // 分子系数
    float a1, a2;      // 分母系数 (a0=1)
    float x1, x2;      // 输入延迟
    float y1, y2;      // 输出延迟
} NotchFilter2ndOrder;

float notch_filter_process(NotchFilter2ndOrder* filter, float input) {
    // 直接形式II实现
    float output = filter->b0 * input + filter->b1 * filter->x1 + filter->b2 * filter->x2
                   - filter->a1 * filter->y1 - filter->a2 * filter->y2;
    
    // 更新延迟
    filter->x2 = filter->x1;
    filter->x1 = input;
    filter->y2 = filter->y1;
    filter->y1 = output;
    
    return output;
}
